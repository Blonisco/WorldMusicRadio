cmake_minimum_required(VERSION 3.15)
project(WorldMusicRadio LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)

list(APPEND CMAKE_PREFIX_PATH "D:/msys64/ucrt64")

# 确保你可以找到所需的库
set(glfw3_DIR "D:/msys64/ucrt64/lib/cmake/glfw3")
set(CPR_DIR "D:/msys64/ucrt64/lib/cmake/cpr")
find_package(glfw3 REQUIRED)
find_package(OpenGL REQUIRED)
find_package(CPR REQUIRED)

include_directories(D:/msys64/urct64/include)
link_directories(D:/msys64/urct64/lib)

set(IMGUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/imgui")
set(Src_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src")
set(Target_DIR "${CMAKE_CURRENT_SOURCE_DIR}/target")

# 添加编译命令
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# 定义源文件
set(SOURCES
    ${Src_DIR}/main.cpp
    ${Src_DIR}/UI.cpp
    ${Src_DIR}/RadioPlayer.cpp
    ${Src_DIR}/apiProvider.cpp
    ${IMGUI_DIR}/imgui.cpp
    ${IMGUI_DIR}/imgui_draw.cpp
    ${IMGUI_DIR}/imgui_tables.cpp
    ${IMGUI_DIR}/imgui_widgets.cpp
    ${IMGUI_DIR}/backends/imgui_impl_glfw.cpp    
    ${IMGUI_DIR}/backends/imgui_impl_opengl3.cpp
)

# 创建可执行文件
add_executable(${PROJECT_NAME} WIN32  ${SOURCES})

# 指定额外的头文件路径
target_include_directories(${PROJECT_NAME} PRIVATE
    ${IMGUI_DIR}
    ${IMGUI_DIR}/backends
)

# 链接库，确保链接到 CPR、GLFW、OpenGL 和 VLC
target_link_libraries(${PROJECT_NAME} PRIVATE
    glfw
    OpenGL::GL
    vlc
    cpr::cpr
)
